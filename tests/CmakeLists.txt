cmake_minimum_required(VERSION 3.21.2 FATAL_ERROR)

# Global Options
set_property(GLOBAL PROPERTY USE_FOLDERS ON) # Group the default targets in a folder


project(tests
	DESCRIPTION "Test project for Synodic Template Library"
	LANGUAGES CXX
)


add_executable(tests)

set_target_properties(tests
	PROPERTIES 
		LINKER_LANGUAGE CXX
		CXX_STANDARD 23
		CXX_STANDARD_REQUIRED ON
		CXX_EXTENSIONS NO
)

find_package(ut 1.1.8 REQUIRED)

target_link_libraries(tests
PRIVATE
	${CMAKE_PROJECT_NAME}
    boost::ut
)

add_subdirectory(src)
